/** * 台灣零件庫存管控系統 PRO v2.5 - Google Apps Script API */ function doGet(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheetName = (e && e.parameter && e.parameter.sheet) ? e.parameter.sheet : "Stock in TW"; let sheet = ss.getSheetByName(sheetName) || ss.getSheets()[0]; const data = sheet.getDataRange().getValues(); if (data.length < 2) return responseJSON({ status: "success", data: [] }); const headers = data[0].map(h => String(h).trim()); const rows = data.slice(1); const result = rows.map((row, index) => { let obj = { id: index + 1 }; headers.forEach((header, colIdx) => { if (!header) return; let val = row[colIdx]; if (val instanceof Date) val = formatDate(val); else if (typeof val === 'number' && val > 40000 && val < 60000) val = formatExcelDate(val); const h = header.toLowerCase(); if (h.includes('location')) obj.location = val || 'TW'; else if (h.includes('parts name')) obj.partsName = val || ''; else if (h.includes('model')) obj.model = val || ''; else if (h === 'qty') obj.qty = Number(val || 0); else if (h === 'qty_1') obj.qty1 = Number(val || 0); else if (h === 'in') obj.in = Number(val || 0); else if (h === 'out') obj.out = Number(val || 0); else if (h.includes('date')) obj.date = val || ''; else if (h.includes('remark')) obj.remark = val || ''; }); return obj; }); return responseJSON({ status: "success", total: result.length, data: result }); } catch (err) { return responseJSON({ status: "error", message: err.toString() }); } } function doPost(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); if (!e.postData || !e.postData.contents) return responseJSON({ status: "error", message: "No post contents" }); const payload = JSON.parse(e.postData.contents); let sheet = ss.getSheetByName(payload.targetSheet || "Stock in TW"); if (!sheet) { sheet = ss.insertSheet(payload.targetSheet || "Stock in TW"); sheet.appendRow(["LOCATION", "PARTS NAME", "MODEL", "QTY", "QTY_1", "IN", "OUT", "DATE", "REMARK"]); } const newRow = [ payload.location || "S6", payload.partsName || "", payload.model || "", Number(payload.qty || 0), Number(payload.qty1 !== undefined ? payload.qty1 : payload.qty || 0), Number(payload.in || 0), Number(payload.out || 0), payload.date || formatDate(new Date()), payload.remark || "" ]; sheet.appendRow(newRow); return responseJSON({ status: "success", insertedRow: newRow }); } catch (err) { return responseJSON({ status: "error", message: err.toString() }); } } function responseJSON(data) { return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON); } function formatDate(d) { return `${d.getFullYear()}/${String(d.getMonth()+1).padStart(2,'0')}/${String(d.getDate()).padStart(2,'0')}`; } function formatExcelDate(serial) { return formatDate(new Date((serial - 25569) * 86400 * 1000)); }
